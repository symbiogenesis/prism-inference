cmake_minimum_required(VERSION 3.24)
project(SwiftPrismRustNativeTest LANGUAGES C CXX)
if(NOT DEFINED PRISM_SOURCE_DIR)
  message(FATAL_ERROR "PRISM_SOURCE_DIR must name the verified, overlaid pinned source")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PRISM_SOURCE_DIR}/ggml" "${CMAKE_CURRENT_BINARY_DIR}/ggml")
# Cargo builds the Rust regression against these pinned external libraries.
enable_testing()
if(BONSAI_PRISM_NATIVE_TEST)
  add_test(NAME prism-lora-storage-alias
    COMMAND "${BONSAI_PRISM_NATIVE_TEST}" --exact native_storage_and_math)
  add_test(NAME prism-lora-loader-rejections
    COMMAND "${BONSAI_PRISM_NATIVE_TEST}" --exact native_loader_rejects_invalid_reads_and_keeps_fallbacks)
endif()
